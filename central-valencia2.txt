    <script>
        // Control de la Intro (Desaparición automática tras 7 segundos)
        window.addEventListener('load', function() {
            setTimeout(function() {
                const intro = document.getElementById('introScreen');
                intro.style.opacity = '0';
                setTimeout(function() {
                    intro.style.display = 'none';
                }, 1000); 
            }, 7000); 
        });

        // CREDENCIALES
        const USER_SECRETARIA = "secretaria";
        const PASSWORD_SECRETARIA = "secre2026";
        const PASSWORD_DEPTO = "valencia123";

        const menuPrincipal = document.getElementById('menuPrincipal');
        const vistaSecretaria = document.getElementById('vistaSecretaria');
        const subtituloPrincipal = document.getElementById('subtituloPrincipal');
        
        const btnVolver = document.getElementById('btnVolver');
        const modalLoginSecretaria = document.getElementById('modalLoginSecretaria');
        const modalGenerico = document.getElementById('modalGenerico');
        const tituloModalDepto = document.getElementById('tituloModalDepto');
        const deptoActualInput = document.getElementById('deptoActual');
        const indiceEdicionInput = document.getElementById('indiceEdicion');
        const formGenerico = document.getElementById('formGenerico');
        const tablaControlSecretaria = document.getElementById('tablaControlSecretaria');
        
        const seccionTablas = document.getElementById('seccionTablas');
        const seccionArchivos = document.getElementById('seccionArchivos');
        const tablaArchivos = document.getElementById('tablaArchivos');

        let departamentoSeleccionadoSecretaria = 'ujieres';

        function abrirLoginSecretaria() {
            modalLoginSecretaria.style.display = 'flex';
        }

        function cerrarLoginSecretaria() {
            modalLoginSecretaria.style.display = 'none';
            document.getElementById('formLoginSecretaria').reset();
        }

        function verificarLoginSecretaria(event) {
            event.preventDefault();
            const usuarioIngresado = document.getElementById('usuarioSecretaria').value;
            const passwordIngresado = document.getElementById('passwordSecretaria').value;

            if (usuarioIngresado === USER_SECRETARIA && passwordIngresado === PASSWORD_SECRETARIA) {
                cerrarLoginSecretaria();
                menuPrincipal.style.display = 'none';
                subtituloPrincipal.innerText = 'Gestión y Control de Departamentos';
                cambiarVistaSecretaria('ujieres', document.querySelector('.sec-tab-btn'));
                vistaSecretaria.style.display = 'block';
            } else {
                alert("Usuario o contraseña incorrectos.");
            }
        }

        function solicitarPasswordDepto(claveDepto, nombreDepto) {
            let pass = prompt(`Introduce la contraseña para el departamento de ${nombreDepto}:`);
            if (pass === PASSWORD_DEPTO) {
                abrirFormulario(claveDepto, nombreDepto);
            } else if (pass !== null) {
                alert("Contraseña incorrecta.");
            }
        }

        btnVolver.addEventListener('click', () => {
            vistaSecretaria.style.display = 'none';
            subtituloPrincipal.innerText = 'Panel de Acceso y Departamentos';
            menuPrincipal.style.display = 'flex';
        });

        window.addEventListener('click', (event) => {
            if (event.target === modalLoginSecretaria) cerrarLoginSecretaria();
            if (event.target === modalGenerico) modalGenerico.style.display = 'none';
        });

        function abrirFormulario(claveDepto, nombreDepto, indice = -1, datos = null) {
            deptoActualInput.value = claveDepto;
            indiceEdicionInput.value = indice;

            if (datos) {
                tituloModalDepto.innerText = `Editar Registro - ${nombreDepto}`;
                document.getElementById('gen-nombre').value = datos.nombre;
                document.getElementById('gen-apellido').value = datos.apellido;
                document.getElementById('gen-telefono').value = datos.telefono;
                document.getElementById('gen-ciudad').value = datos.ciudad;
                document.getElementById('btnGuardarForm').innerText = 'Actualizar Registro';
            } else {
                tituloModalDepto.innerText = `Registro - ${nombreDepto}`;
                formGenerico.reset();
                document.getElementById('btnGuardarForm').innerText = 'Enviar Registro';
            }
            modalGenerico.style.display = 'flex';
        }

        function cerrarFormulario() {
            modalGenerico.style.display = 'none';
            formGenerico.reset();
            indiceEdicionInput.value = -1;
        }

        formGenerico.addEventListener('submit', function(event) {
            event.preventDefault();
            const depto = deptoActualInput.value;
            const indice = parseInt(indiceEdicionInput.value);

            const registroActualizado = {
                nombre: document.getElementById('gen-nombre').value,
                apellido: document.getElementById('gen-apellido').value,
                telefono: document.getElementById('gen-telefono').value,
                ciudad: document.getElementById('gen-ciudad').value
            };

            let storageKey = 'reg_' + depto;
            let registros = JSON.parse(localStorage.getItem(storageKey)) || [];

            if (indice === -1) {
                registros.push(registroActualizado);
                alert('¡Registro guardado con éxito!');
            } else {
                registros[indice] = registroActualizado;
                alert('¡Registro actualizado con éxito!');
            }

            localStorage.setItem(storageKey, JSON.stringify(registros));
            cerrarFormulario();

            if (vistaSecretaria.style.display === 'block' && departamentoSeleccionadoSecretaria !== 'archivos') {
                cargarDatosSecretaria(depto);
            }
        });

        function cambiarVistaSecretaria(depto, elementoTab) {
            departamentoSeleccionadoSecretaria = depto;
            document.querySelectorAll('.sec-tab-btn').forEach(btn => btn.classList.remove('active'));
            elementoTab.classList.add('active');

            if (depto === 'archivos') {
                seccionTablas.style.display = 'none';
                seccionArchivos.style.display = 'block';
                cargarArchivos();
            } else {
                seccionArchivos.style.display = 'none';
                seccionTablas.style.display = 'block';
                cargarDatosSecretaria(depto);
            }
        }

        function cargarDatosSecretaria(depto) {
            let storageKey = 'reg_' + depto;
            let registros = JSON.parse(localStorage.getItem(storageKey)) || [];
            tablaControlSecretaria.innerHTML = '';

            if (registros.length === 0) {
                tablaControlSecretaria.innerHTML = `<tr><td colspan="5" class="no-data">No hay registros en este departamento.</td></tr>`;
                return;
            }

            let nombresDeptos = {
                'ujieres': 'Ujieres',
                'escuela-dominical': 'Escuela Dominical',
                'caballeros': 'Caballeros',
                'mujeres-del-bien': 'Mujeres del Bien',
                'jovenes': 'Jóvenes',
                'alabanza': 'Alabanza'
            };

            registros.forEach((reg, index) => {
                let fila = document.createElement('tr');
                let datosJson = JSON.stringify(reg).replace(/"/g, '&quot;');
                fila.innerHTML = `
                    <td>${reg.nombre}</td>
                    <td>${reg.apellido}</td>
                    <td>${reg.telefono}</td>
                    <td>${reg.ciudad}</td>
                    <td>
                        <div class="btn-acciones">
                            <button class="btn-editar" onclick="prepararEdicion('${depto}', '${nombresDeptos[depto]}', ${index}, ${datosJson})">Editar</button>
                            <button class="btn-borrar" onclick="eliminarRegistroSecretaria('${depto}', ${index})">Eliminar</button>
                        </div>
                    </td>
                `;
                tablaControlSecretaria.appendChild(fila);
            });
        }

        function prepararEdicion(depto, nombreDepto, index, datos) {
            abrirFormulario(depto, nombreDepto, index, datos);
        }

        function eliminarRegistroSecretaria(depto, index) {
            if (confirm('¿Estás seguro de que deseas eliminar este registro?')) {
                let storageKey = 'reg_' + depto;
                let registros = JSON.parse(localStorage.getItem(storageKey)) || [];
                registros.splice(index, 1);
                localStorage.setItem(storageKey, JSON.stringify(registros));
                cargarDatosSecretaria(depto);
            }
        }

        function subirArchivo(event) {
            event.preventDefault();
            const nombreArchivo = document.getElementById('nombreArchivo').value;
            const archivoInput = document.getElementById('archivoInput').files[0];

            if (!archivoInput) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const archivoData = {
                    nombre: nombreArchivo,
                    tipo: archivoInput.name.split('.').pop().toUpperCase(),
                    url: e.target.result
                };

                let archivos = JSON.parse(localStorage.getItem('archivosSecretaria')) || [];
                archivos.push(archivoData);
                localStorage.setItem('archivosSecretaria', JSON.stringify(archivos));

                alert('¡Archivo subido con éxito!');
                document.getElementById('formArchivo').reset();
                cargarArchivos();
            };
            reader.readAsDataURL(archivoInput);
        }

        function cargarArchivos() {
            let archivos = JSON.parse(localStorage.getItem('archivosSecretaria')) || [];
            tablaArchivos.innerHTML = '';

            if (archivos.length === 0) {
                tablaArchivos.innerHTML = `<tr><td colspan="3" class="no-data">No hay archivos subidos todavía.</td></tr>`;
                return;
            }

            archivos.forEach((arch, index) => {
                let fila = document.createElement('tr');
                fila.innerHTML = `
                    <td>${arch.nombre}</td>
                    <td>${arch.tipo}</td>
                    <td>
                        <div class="btn-acciones">
                            <a href="${arch.url}" download="${arch.nombre}.${arch.tipo.toLowerCase()}" class="btn-editar" style="text-align:center;">Descargar</a>
                            <button class="btn-borrar" onclick="eliminarArchivo(${index})">Eliminar</button>
                        </div>
                    </td>
                `;
                tablaArchivos.appendChild(fila);
            });
        }

        function eliminarArchivo(index) {
            if (confirm('¿Estás seguro de que deseas eliminar este archivo?')) {
                let archivos = JSON.parse(localStorage.getItem('archivosSecretaria')) || [];
                archivos.splice(index, 1);
